cmake_minimum_required(VERSION 3.24)
project(MediaWaveThumbnails VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
  add_compile_definitions(NOMINMAX)
endif()
if(MSVC)
  add_compile_options(/utf-8)
endif()
include(CTest)
find_package(Threads REQUIRED)
option(MWT_BUILD_PLUGIN "Build the REAPER extension" ON)
set(MWT_WDL_DIR "" CACHE PATH "Existing checkout of justinfrankel/WDL (optional)")
if(MWT_BUILD_PLUGIN)
  if(WIN32)
    add_library(reaper_media_wave_thumbnails SHARED src/plugin.cpp)
    target_compile_definitions(reaper_media_wave_thumbnails PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(reaper_media_wave_thumbnails PRIVATE comctl32 gdi32 user32 Threads::Threads)
    if(MINGW)
      target_link_options(reaper_media_wave_thumbnails PRIVATE -static -pthread)
    endif()
  elseif(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 17)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    if(NOT MWT_WDL_DIR)
      include(FetchContent)
      FetchContent_Declare(mwt_wdl GIT_REPOSITORY https://github.com/justinfrankel/WDL.git
        GIT_TAG 03c6e564b881a41cb13441cfd7350e92ae1ddf4d SOURCE_SUBDIR _mwt_headers_only)
      FetchContent_MakeAvailable(mwt_wdl)
      set(MWT_WDL_DIR "${mwt_wdl_SOURCE_DIR}")
    endif()
    if(NOT EXISTS "${MWT_WDL_DIR}/WDL/swell/swell-modstub.mm")
      message(FATAL_ERROR "MWT_WDL_DIR must be a WDL repository checkout")
    endif()
    # Preserve the official SDK's relative ../WDL/swell/swell.h include.
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/sdk/reaper")
    configure_file(third_party/reaper/reaper_plugin.h "${CMAKE_CURRENT_BINARY_DIR}/sdk/reaper/reaper_plugin.h" COPYONLY)
    file(CREATE_LINK "${MWT_WDL_DIR}/WDL" "${CMAKE_CURRENT_BINARY_DIR}/sdk/WDL" SYMBOLIC COPY_ON_ERROR)
    add_library(reaper_media_wave_thumbnails SHARED src/mac_plugin.mm "${MWT_WDL_DIR}/WDL/swell/swell-modstub.mm")
    target_include_directories(reaper_media_wave_thumbnails PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/sdk")
    target_compile_definitions(reaper_media_wave_thumbnails PRIVATE SWELL_PROVIDED_BY_APP NOMINMAX)
    target_compile_options(reaper_media_wave_thumbnails PRIVATE -fno-objc-arc -Wno-deprecated-declarations)
    target_link_libraries(reaper_media_wave_thumbnails PRIVATE "-framework Cocoa" Threads::Threads)
    set_target_properties(reaper_media_wave_thumbnails PROPERTIES SUFFIX ".dylib")
  else()
    message(FATAL_ERROR "Native UI backend is implemented only for Windows/macOS; use -DMWT_BUILD_PLUGIN=OFF for portable tests")
  endif()
  set_target_properties(reaper_media_wave_thumbnails PROPERTIES PREFIX "")
  install(TARGETS reaper_media_wave_thumbnails LIBRARY DESTINATION . RUNTIME DESTINATION .)
endif()
if(BUILD_TESTING)
  add_executable(portable_core_test tests/portable-core.cpp)
  target_link_libraries(portable_core_test PRIVATE Threads::Threads)
  add_test(NAME portable_core COMMAND portable_core_test "${CMAKE_CURRENT_BINARY_DIR}/portable-test-data")
endif()
